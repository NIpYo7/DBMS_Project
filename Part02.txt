create table Lecturer(
Lecturer_id varchar(15) not null primary key,
F_Name char(150),
L_Name char(150),
Gender char(10),
Contact Number(15),
Email varchar(50)
);

create table Department(
Dep_id varchar(20) not null primary key,
Name char(150),
Location varchar(200),
Dean_ID varchar(20),
Admin_ID varchar(20)
);

CREATE TABLE Technical_Officer (
    TO_ID VARCHAR(25) NOT NULL PRIMARY KEY,
    Email VARCHAR(30),
    F_Name CHAR(60),
    L_Name CHAR(60),
    Gender CHAR(15),
    Contact_Number VARCHAR(15),
    Dep_ID varchar(20)
);

create table Attendance(
Attendance_id varchar (25) not null primary key,
Time varchar(15),
Status varchar(20),
Type varchar(50),
Hours int(50),
Date date,
Course_ID varchar(20)
);
